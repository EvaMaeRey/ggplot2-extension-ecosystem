pkgs <- as_tibble(tools::CRAN_package_db())
gg_pkgs <- pkgs %>% 
  mutate(
    ind_ggplot_depend_import_suggest = if_any(-Package, ~ str_detect(.x, "ggplot2")),
    ind_gg = str_detect(Package, "gg|GG"),
    ind_ggplot_in_description = str_detect(Description, "ggplot"),
    ind_extension_in_description = str_detect(Description, "exten") 
         ) %>%
  filter(ind_ggplot_depend_import_suggest, 
         ind_gg | ind_ggplot_in_description) %>% 
  mutate(ext_score = ind_gg + ind_ggplot_in_description + ind_extension_in_description)
